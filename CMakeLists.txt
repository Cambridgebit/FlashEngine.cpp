cmake_minimum_required(VERSION 3.24)
project(FlashEngineNative VERSION 0.4.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "原生部署首版仅支持 Linux；Windows 请在 WSL 内构建")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FE_CUDA "构建 CUDA 模型执行后端" OFF)
option(FE_COREX "构建 CoreX 兼容 LibTorch C++ 后端" OFF)
option(FE_ASCEND "构建纯 AscendCL/ACLNN 后端" OFF)
option(FE_REFERENCE_OPS "CUDA 上编译可移植参考算子，用于 CoreX 源码回归" OFF)
option(FE_ASAN "启用主机地址/未定义行为检查" OFF)
option(FE_DEVICE_TESTS "启用需要真实设备的算子测试" OFF)
if((FE_CUDA AND FE_COREX) OR (FE_CUDA AND FE_ASCEND) OR (FE_COREX AND FE_ASCEND))
  message(FATAL_ERROR "每个部署物只能链接一个厂商运行时；请用独立构建目录")
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost 1.74 REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc)
find_path(PCRE2_INCLUDE_DIR pcre2.h REQUIRED)
find_library(PCRE2_LIBRARY pcre2-8 REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
set(FE_MINJA_DIR "${CMAKE_SOURCE_DIR}/third_party/minja/include" CACHE PATH "固定提交的 Minja 头文件")
if(NOT EXISTS "${FE_MINJA_DIR}/minja/minja.hpp")
  message(FATAL_ERROR "缺少固定版本 Minja；请先按 README 准备离线依赖")
endif()
add_library(fe_host src/common.cpp src/bundle.cpp src/tokenizer.cpp src/scheduler.cpp src/profiles.cpp)
target_include_directories(fe_host PUBLIC include ${JSON_INCLUDE_DIR} PRIVATE ${PCRE2_INCLUDE_DIR} ${FE_MINJA_DIR})
target_link_libraries(fe_host PUBLIC OpenSSL::Crypto Threads::Threads ${PCRE2_LIBRARY} ICU::uc)
target_compile_options(fe_host PRIVATE -Wall -Wextra -Wpedantic)
if(FE_ASAN)
  target_compile_options(fe_host PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(fe_host PUBLIC -fsanitize=address,undefined)
endif()
add_executable(flashengine-native src/main.cpp)
target_link_libraries(flashengine-native PRIVATE fe_host)
add_library(fe_serving src/engine.cpp src/server.cpp src/cli.cpp src/benchmark.cpp)
target_include_directories(fe_serving PUBLIC include)
target_include_directories(fe_serving SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(fe_serving PUBLIC fe_host)
if(FE_CUDA)
  if(NOT FE_REFERENCE_OPS)
    enable_language(CUDA)
  endif()
  find_package(Torch REQUIRED)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
  set(FE_FLASHINFER_INCLUDE_DIR "" CACHE PATH "固定版本 FlashInfer 原生头文件目录")
  if(NOT FE_REFERENCE_OPS AND NOT EXISTS "${FE_FLASHINFER_INCLUDE_DIR}/flashinfer/attention/prefill.cuh")
    message(FATAL_ERROR "CUDA 构建需要 FE_FLASHINFER_INCLUDE_DIR")
  endif()
  if(FE_REFERENCE_OPS)
    add_library(fe_cuda src/cuda_model.cpp src/hybrid_model.cpp src/block_draft.cpp src/portable_gpu_ops.cpp)
    target_compile_definitions(fe_cuda PRIVATE FE_ATTENTION_ID="aten-paged-reference")
    target_compile_definitions(flashengine-native PRIVATE FE_REFERENCE_OPS_BUILD=1)
  else()
    add_library(fe_cuda src/cuda_model.cpp src/hybrid_model.cpp src/block_draft.cpp src/cuda_kernels.cu src/flashinfer_attention.cu)
  endif()
  target_include_directories(fe_cuda PUBLIC include ${TORCH_INCLUDE_DIRS})
  target_link_libraries(fe_cuda PUBLIC fe_host fe_serving ${TORCH_LIBRARIES} CUDA::cudart)
  target_include_directories(fe_cuda PRIVATE ${FE_FLASHINFER_INCLUDE_DIR})
  target_compile_options(fe_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--extended-lambda>)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89)
  endif()
  set_target_properties(fe_cuda PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
  target_compile_definitions(flashengine-native PRIVATE FE_HAS_CUDA=1 FE_DEFAULT_BACKEND="cuda")
  target_link_libraries(flashengine-native PRIVATE fe_cuda)
endif()
if(FE_COREX)
  set(FE_COREX_ROOT "/usr/local/corex" CACHE PATH "CoreX SDK 根目录")
  set(FE_COREX_TORCH_ROOT "" CACHE PATH "厂商 LibTorch 根目录，必须含 include 和 lib")
  set(FE_TORCH_CXX11_ABI "" CACHE STRING "厂商 LibTorch 的 _GLIBCXX_USE_CXX11_ABI（0 或 1）")
  if(NOT EXISTS "${FE_COREX_TORCH_ROOT}/include/ATen/ATen.h" OR NOT EXISTS "${FE_COREX_ROOT}/include/cuda_runtime.h" OR NOT FE_TORCH_CXX11_ABI MATCHES "^[01]$")
    message(FATAL_ERROR "CoreX 需要厂商 LibTorch C++、SDK 和明确的 FE_TORCH_CXX11_ABI；不能使用 NVIDIA LibTorch 代替")
  endif()
  add_library(fe_corex src/cuda_model.cpp src/hybrid_model.cpp src/block_draft.cpp src/portable_gpu_ops.cpp)
  target_include_directories(fe_corex PRIVATE "${FE_COREX_TORCH_ROOT}/include" "${FE_COREX_TORCH_ROOT}/include/torch/csrc/api/include" "${FE_COREX_ROOT}/include")
  target_compile_definitions(fe_corex PRIVATE _GLIBCXX_USE_CXX11_ABI=${FE_TORCH_CXX11_ABI} FE_GPU_BACKEND="corex" FE_ATTENTION_ID="aten-paged-reference")
  # ABI 必须传递到所有跨 LibTorch 边界的 C++ 类型。
  target_compile_definitions(fe_host PUBLIC _GLIBCXX_USE_CXX11_ABI=${FE_TORCH_CXX11_ABI})
  foreach(lib torch torch_cpu torch_cuda c10 c10_cuda)
    find_library(FE_COREX_${lib} NAMES ${lib} PATHS "${FE_COREX_TORCH_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(fe_corex PRIVATE ${FE_COREX_${lib}})
  endforeach()
  find_library(FE_COREX_CUDART NAMES cudart PATHS "${FE_COREX_ROOT}/lib64" "${FE_COREX_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(fe_corex PUBLIC fe_serving PRIVATE ${FE_COREX_CUDART})
  target_compile_definitions(flashengine-native PRIVATE FE_HAS_CUDA=1 FE_DEFAULT_BACKEND="corex")
  target_link_libraries(flashengine-native PRIVATE fe_corex)
endif()
if(FE_ASCEND)
  set(FE_ASCEND_ROOT "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN Toolkit 根目录")
  set(FE_ATB_ROOT "/usr/local/Ascend/nnal/atb/9.0.0/atb/cxx_abi_1" CACHE PATH "ATB 原生 C++11 ABI 开发目录")
  find_path(FE_ATB_INCLUDE atb/atb_infer.h PATHS "${FE_ATB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(FE_ATB_LIBRARY NAMES atb PATHS "${FE_ATB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  find_path(FE_ACL_INCLUDE acl/acl.h PATHS "${FE_ASCEND_ROOT}/include" "${FE_ASCEND_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include" NO_DEFAULT_PATH REQUIRED)
  find_path(FE_ACLNN_INCLUDE aclnnop/aclnn_rms_norm.h PATHS "${FE_ASCEND_ROOT}/include" "${FE_ASCEND_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include" "${FE_ASCEND_ROOT}/opp/built-in/op_impl/ai_core/tbe/op_api/include" NO_DEFAULT_PATH REQUIRED)
  foreach(lib ascendcl nnopbase opapi)
    find_library(FE_ASCEND_${lib} NAMES ${lib} PATHS "${FE_ASCEND_ROOT}/lib64" "${FE_ASCEND_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/lib64" "${FE_ASCEND_ROOT}/opp/built-in/op_impl/ai_core/tbe/op_api/lib" NO_DEFAULT_PATH REQUIRED)
  endforeach()
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_INCLUDES "${FE_ACL_INCLUDE};${FE_ACLNN_INCLUDE};${FE_ATB_INCLUDE};${CMAKE_SOURCE_DIR}/include;${JSON_INCLUDE_DIR}")
  set(CMAKE_REQUIRED_LIBRARIES "${FE_ASCEND_ascendcl};${FE_ASCEND_nnopbase};${FE_ASCEND_opapi};${FE_ATB_LIBRARY}")
  check_cxx_source_compiles("#include \"${CMAKE_SOURCE_DIR}/src/ascend_ops.hpp\"\nint main(){volatile auto begin=&aclmdlRICaptureBegin;volatile auto reuse=&aclSetAclOpExecutorRepeatable;volatile auto destroy=&aclDestroyAclOpExecutor;return begin&&reuse&&destroy?0:1;}" FE_ASCEND_PUBLIC_API_OK)
  unset(CMAKE_REQUIRED_INCLUDES)
  unset(CMAKE_REQUIRED_LIBRARIES)
  if(NOT FE_ASCEND_PUBLIC_API_OK)
    message(FATAL_ERROR "CANN 缺少所需 ACLNN/Graph/可复用 executor API，或 API 原型不匹配；查看 CMakeConfigureLog.yaml，禁止回退 TorchNPU 私有符号")
  endif()
  add_library(fe_ascend src/ascend_model.cpp)
  target_include_directories(fe_ascend PRIVATE ${FE_ACL_INCLUDE} ${FE_ACLNN_INCLUDE} ${FE_ATB_INCLUDE})
  target_link_libraries(fe_ascend PUBLIC fe_serving PRIVATE ${FE_ASCEND_ascendcl} ${FE_ASCEND_nnopbase} ${FE_ASCEND_opapi} ${FE_ATB_LIBRARY} ${CMAKE_DL_LIBS})
  target_compile_definitions(flashengine-native PRIVATE FE_HAS_ASCEND=1)
  target_link_libraries(flashengine-native PRIVATE fe_ascend)
endif()
